import pandas as pd
import numpy as np
import random
from datetime import datetime, timedelta

# Set random seed for reproducibility
np.random.seed(42)

# Generate the dataset
data = []
coffee_types = ['Espresso', 'Latte', 'Cappuccino', 'Americano', 'Cold Brew', 'Mocha', 'Flat White', 'Macchiato']
size_options = ['Small', 'Medium', 'Large']
loyalty_tiers = ['Bronze', 'Silver', 'Gold', 'Platinum']
cafes = ['Downtown Brew', 'Central Perk', 'Bean There', 'Urban Grind', 'Morning Buzz']

start_date = datetime(2024, 1, 1)

for i in range(5001):
    customer_id = f"CUST{1000 + i}"
    age = np.random.randint(18, 75)
    visits_per_week = np.random.poisson(3.5)
    favorite_drink = random.choice(coffee_types)
    usual_size = random.choice(size_options)
    
    # Generate some purchase history
    total_spent = round(np.random.normal(150, 50), 2)
    if total_spent < 0:
        total_spent = 10.0
    
    # Loyalty tier based on spending
    if total_spent > 250:
        loyalty_tier = 'Platinum'
    elif total_spent > 180:
        loyalty_tier = 'Gold'
    elif total_spent > 120:
        loyalty_tier = 'Silver'
    else:
        loyalty_tier = 'Bronze'
    
    # Preferences
    prefers_organic = random.choice([True, False])
    prefers_plant_milk = random.choice([True, False])
    favorite_cafe = random.choice(cafes)
    
    # Generate last visit date (within last 90 days)
    days_ago = np.random.randint(0, 90)
    last_visit = start_date - timedelta(days=days_ago)
    
    # Satisfaction score (1-10)
    satisfaction = np.random.randint(6, 11)
    
    data.append({
        'customer_id': customer_id,
        'age': age,
        'visits_per_week': visits_per_week,
        'favorite_drink': favorite_drink,
        'usual_size': usual_size,
        'total_spent': total_spent,
        'loyalty_tier': loyalty_tier,
        'prefers_organic': prefers_organic,
        'prefers_plant_milk': prefers_plant_milk,
        'favorite_cafe': favorite_cafe,
        'last_visit': last_visit.strftime('%Y-%m-%d'),
        'satisfaction_score': satisfaction
    })

# Create DataFrame
df = pd.DataFrame(data)

# Save to CSV
df.to_csv('coffee_shop_customers.csv', index=False)

print("Dataset created with 5001 entries!")
print(f"Dataset shape: {df.shape}")
print("\nFirst few entries:")
print(df.head())
print("\nDataset info:")
print(df.info())

# coffee_shop_analysis.py
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from datetime import datetime

# Load the dataset
df = pd.read_csv('coffee_shop_customers.csv')

print("=== COFFEE SHOP CUSTOMER ANALYSIS ===\n")

# Basic information
print("1. DATASET OVERVIEW")
print(f"Total customers: {len(df):,}")
print(f"Dataset shape: {df.shape}")
print("\nFirst 5 entries:")
print(df.head())

print("\n2. BASIC STATISTICS")
print(df.describe())

print("\n3. CATEGORICAL DATA SUMMARY")
categorical_cols = ['favorite_drink', 'usual_size', 'loyalty_tier', 'favorite_cafe']
for col in categorical_cols:
    print(f"\n{col.upper()} distribution:")
    print(df[col].value_counts())

print("\n4. LOYALTY ANALYSIS")
loyalty_stats = df.groupby('loyalty_tier').agg({
    'total_spent': ['mean', 'median', 'count'],
    'satisfaction_score': 'mean',
    'visits_per_week': 'mean'
}).round(2)
print(loyalty_stats)

print("\n5. CUSTOMER PREFERENCES")
print(f"Customers who prefer organic: {df['prefers_organic'].sum():,} ({(df['prefers_organic'].mean()*100):.1f}%)")
print(f"Customers who prefer plant milk: {df['prefers_plant_milk'].sum():,} ({(df['prefers_plant_milk'].mean()*100):.1f}%)")

# Visualization setup
plt.style.use('default')
fig, axes = plt.subplots(2, 3, figsize=(18, 12))
fig.suptitle('Coffee Shop Customer Analysis', fontsize=16)

# Plot 1: Favorite drinks
drink_counts = df['favorite_drink'].value_counts()
axes[0,0].bar(drink_counts.index, drink_counts.values)
axes[0,0].set_title('Favorite Drink Distribution')
axes[0,0].tick_params(axis='x', rotation=45)

# Plot 2: Loyalty tiers
loyalty_counts = df['loyalty_tier'].value_counts()
axes[0,1].pie(loyalty_counts.values, labels=loyalty_counts.index, autopct='%1.1f%%')
axes[0,1].set_title('Loyalty Tier Distribution')

# Plot 3: Age distribution
axes[0,2].hist(df['age'], bins=20, edgecolor='black', alpha=0.7)
axes[0,2].set_title('Age Distribution')
axes[0,2].set_xlabel('Age')
axes[0,2].set_ylabel('Count')

# Plot 4: Spending by loyalty tier
df.boxplot(column='total_spent', by='loyalty_tier', ax=axes[1,0])
axes[1,0].set_title('Spending by Loyalty Tier')
axes[1,0].set_xlabel('Loyalty Tier')

# Plot 5: Visits per week
visit_counts = df['visits_per_week'].value_counts().sort_index()
axes[1,1].bar(visit_counts.index, visit_counts.values)
axes[1,1].set_title('Visits Per Week Distribution')
axes[1,1].set_xlabel('Visits per Week')

# Plot 6: Satisfaction scores
satisfaction_counts = df['satisfaction_score'].value_counts().sort_index()
axes[1,2].bar(satisfaction_counts.index, satisfaction_counts.values)
axes[1,2].set_title('Satisfaction Score Distribution')
axes[1,2].set_xlabel('Satisfaction Score')

plt.tight_layout()
plt.savefig('coffee_shop_analysis.png', dpi=300, bbox_inches='tight')
plt.show()

print("\n6. CORRELATION ANALYSIS")
numeric_cols = ['age', 'visits_per_week', 'total_spent', 'satisfaction_score']
correlation_matrix = df[numeric_cols].corr()
print(correlation_matrix)

print("\n7. INSIGHTS AND RECOMMENDATIONS")
print(f"- Most popular drink: {df['favorite_drink'].mode().iloc[0]}")
print(f"- Average customer age: {df['age'].mean():.1f} years")
print(f"- Average weekly visits: {df['visits_per_week'].mean():.2f}")
print(f"- Average satisfaction score: {df['satisfaction_score'].mean():.2f}/10")
print(f"- Total revenue represented: ${df['total_spent'].sum():,.2f}")

# Advanced analysis
print("\n8. ADVANCED ANALYSIS")
# Customer segments based on age and spending
df['age_group'] = pd.cut(df['age'], bins=[18, 25, 35, 50, 75], labels=['18-25', '26-35', '36-50', '51+'])
age_group_analysis = df.groupby('age_group').agg({
    'total_spent': 'mean',
    'visits_per_week': 'mean',
    'satisfaction_score': 'mean'
}).round(2)
print("Age Group Analysis:")
print(age_group_analysis)

print("\nAnalysis complete! Check 'coffee_shop_analysis.png' for visualizations.")
